\chapter{Thiết kế hệ thống và thử nghiệm}
\label{chap:chapter4}

Chương này trình bày yêu cầu hệ thống, kiến trúc triển khai, và kết quả thử nghiệm
trên hai nghiên cứu trường hợp: pháp luật Việt Nam (400 câu hỏi) và quy định đào tạo
(130 câu hỏi), cùng phân tích ablation đánh giá đóng góp của từng thành phần.

\section{Yêu cầu hệ thống}
\label{sec:yeu-cau-he-thong}

\subsection{Yêu cầu chức năng}
\label{subsec:yc-chuc-nang}

\begin{enumerate}
  \item \textbf{Tiếp nhận truy vấn:} Người dùng nhập câu hỏi bằng tiếng Việt về quy định đào
  tạo hoặc pháp luật.

  \item \textbf{Truy xuất điều khoản:} Hệ thống trả về danh sách top-K điều khoản liên quan
  nhất, kèm trích dẫn nguồn (tên văn bản, số điều).

  \item \textbf{Sinh câu trả lời:} LLM tổng hợp thông tin từ các điều khoản đã truy xuất thành
  câu trả lời tự nhiên.

  \item \textbf{Hỗ trợ đa miền:} Hệ thống hoạt động trên nhiều lĩnh vực (doanh nghiệp, giao
  thông, đào tạo) với cùng pipeline và tham số.
\end{enumerate}

\subsection{Yêu cầu phi chức năng}
\label{subsec:yc-phi-chuc-nang}

\begin{enumerate}
  \item \textbf{Độ chính xác:} Hit@10 $\geq$ 85\% trên các bộ đánh giá.

  \item \textbf{Thời gian phản hồi:} Dưới 15 giây cho mỗi truy vấn.

  \item \textbf{Khả năng mở rộng:} Hỗ trợ thêm miền mới mà không cần thay đổi kiến trúc core.

  \item \textbf{Khả năng giải thích:} Mỗi kết quả kèm reasoning trace từ quá trình duyệt cây.
\end{enumerate}

\begin{itemize}
  \item \textbf{Pha ngoại tuyến (Offline):}
  \begin{itemize}
    \item Document Parser: phân tích văn bản thành cây cấu trúc
    \item KG Builder: xây dựng Knowledge Graph qua LLM + expert review
    \item Summary Generator: sinh tóm tắt chương và điều
    \item Embedding Indexer: tạo chỉ mục vector cho tìm kiếm ngữ nghĩa
  \end{itemize}

  \item \textbf{Pha trực tuyến (Online):}
  \begin{itemize}
    \item Query Analyzer: phân loại ý định, mở rộng viết tắt, trích xuất thực thể
    \item Tree Traversal Retriever: duyệt cây 3 vòng lặp (Tầng 1)

    \item Answer Generator: sinh câu trả lời từ LLM

\subsection{Chi tiết triển khai}
\label{subsec:chitiet-trienkhay}

\textbf{Document parsing.} Văn bản pháp luật được phân tích bằng bộ regex tùy chỉnh nhận
diện các dấu hiệu phân cấp tiếng Việt (``Phần'', ``Chương'', ``Mục'', ``Điều'', ``Khoản'',
``Điểm'') và xây dựng cây cấu trúc 6 cấp. Các văn bản scan (PDF) của miền giáo dục
(118 trang) được số hóa qua LLM Vision OCR. Tham chiếu chéo được trích xuất qua
pattern matching trên chuỗi trích dẫn điều khoản, tạo ra 168 cạnh liên tài liệu.

\textbf{Knowledge graph.} Trích xuất thực thể và nhận diện quan hệ sử dụng Claude 3.5
Haiku qua Anthropic API, với một lời gọi LLM cho mỗi điều khoản. Chuẩn hóa slug
tiếng Việt loại bỏ dấu và gộp biến thể thanh điệu thành định danh thực thể chuẩn. Bộ
giải quyết thực thể đa bước (exact match, synonym resolution, abbreviation expansion)
khử trùng lặp các thực thể tương đương. Sau trích xuất tự động, bước tinh chỉnh expert-in-the-loop sửa các quan hệ nhận diện sai và bổ sung liên kết chuyên ngành mà LLM bỏ
sót.

\textbf{Online query processing.} Duyệt cây và chọn điều khoản sử dụng Claude Sonnet
4. Sentence embeddings được tính bằng \texttt{paraphrase-multilingual-MiniLM-L12-v2} \cite{reimers2019sbert},
mô hình đa ngôn ngữ 384 chiều. PPR được tính lặp (tối đa 50 vòng, tolerance $10^{-6}$,
damping $\alpha = 0{,}15$). Tất cả phản hồi LLM được cache trong SQLite để đảm bảo tái lập
và tiết kiệm chi phí.

\textbf{Parallel execution.} Hệ thống hỗ trợ thực thi song song với 10 worker đồng thời,
đạt tăng tốc 6,5$\times$ (226s $\rightarrow$ 35s cho 10 truy vấn). SQLite cache sử dụng WAL mode +
timeout 30s cho truy cập đồng thời an toàn.

\section{Kết quả thử nghiệm}
\label{sec:ket-qua-thu-nghiem}

\subsection{Các chỉ số đánh giá}
\label{subsec:chi-so-danh-gia}

Hệ thống được đánh giá bằng hai chỉ số truy xuất tiêu chuẩn~\cite{manning2008ir}:

\textbf{Top-k Hit Rate} đo tỷ lệ truy vấn mà ít nhất một điều khoản liên quan xuất hiện trong
top-k kết quả:
\begin{equation}
  \text{Hit@}k = \frac{1}{|Q|} \sum_{q} \mathbf{1}\!\left[\exists a \in R_q : \text{rank}(a) \leq k\right]
  \label{eq:hitk}
\end{equation}

\textbf{Mean Reciprocal Rank} trung bình hạng nghịch đảo của kết quả liên quan đầu tiên:
\begin{equation}
  \text{MRR} = \frac{1}{|Q|} \sum_{q} \left(\min_{a \in R_q} \text{rank}(a)\right)^{-1}
  \label{eq:mrr}
\end{equation}

Kết quả được báo cáo tại $k \in \{1, 5, 10, 20\}$, với Hit@10 và MRR là chỉ số chính.
Hit@10 được ưu tiên vì trong pipeline RAG, LLM xác nhận ngữ cảnh đã truy xuất, nên
một điều khoản đúng trong mười ứng viên là đủ.

\subsection{Các hệ thống đối sánh}
\label{subsec:he-thong-doi-sanh}

\begin{itemize}
  \item \textbf{BM25}~\cite{robertson2009bm25}: Truy xuất từ vựng với tokenization mức âm tiết.

  \item \textbf{TF-IDF}~\cite{manning2008ir}: Truy xuất dựa trên tần suất từ--nghịch đảo tần suất tài liệu.

  \item \textbf{NaiveRAG}~\cite{reimers2019sbert}: Truy xuất dày đặc, không reranking.

  \item \textbf{LightRAG}~\cite{guo2025lightrag}: Truy xuất hybrid KG (vector + duyệt đồ thị).

  \item \textbf{RAPTOR}~\cite{sarthi2024raptor}: Summary Tree đệ quy với truy xuất embedding.

  \item \textbf{PageIndex}~\cite{zhang2025pageindex}: Tìm kiếm cây phân cấp hai bước hướng dẫn bởi LLM.
\end{itemize}

Tất cả hệ thống sử dụng Sonnet 4 cho các lời gọi LLM và truy xuất từ cùng kho tài
liệu.

\subsection{Case Study 1: Pháp luật Việt Nam}
\label{subsec:casestudy-phapluat}

\textbf{Dữ liệu.} Bộ dữ liệu pháp luật gồm 400 câu hỏi khó (hard-400) thuộc hai lĩnh vực: luật doanh
nghiệp (200 câu, chủ yếu từ Luật 59/2020/QH14 và 9 nghị định) và luật giao thông (200
câu, từ Luật 36/2024/QH15 và 2 nghị định xử phạt). Câu hỏi được thu thập từ các nhóm
tư vấn pháp lý trên Facebook, với mức độ khó cao đòi hỏi suy luận đa bước.

\textbf{Kết quả.} Bảng~\ref{tab:hit-phapluat} trình bày kết quả trên bộ dữ liệu pháp luật 400 câu hỏi.

ra mức tăng 17,7 pp từ vị trí 6--10 so với chỉ 8,6 pp của PageIndex.

Các baseline RAG (NaiveRAG 32,8\%, LightRAG 39,8\%, RAPTOR 41,0\%) thấp
hơn cả BM25 (47,5\%): embeddings đa mục đích thiếu từ vựng pháp lý tiếng Việt, trích

\label{subsec:ablation}

Để đánh giá đóng góp của từng thành phần, thử nghiệm ablation được thực hiện trên
bộ hard-200 (150 doanh nghiệp + 50 giao thông). Bảng~\ref{tab:ablation} trình bày kết quả.

\begin{table}[htbp]
  \centering
  \caption{Kết quả ablation study (Hit@10 \%, trung bình 3 lần chạy)}
  \label{tab:ablation}
  \begin{tabular}{lrr}
    \toprule
    \textbf{Cấu hình} & \textbf{Hit@10} & \textbf{$\Delta$} \\
    \midrule
    Full (3-Tier + RRF)                    & 88{,}0 & --- \\
    Tree only (Tầng 1)                     & 64{,}8 & $-23{,}2$ \\
    Dual only (Tầng 2)                     & 33{,}0 & $-55{,}0$ \\
    Simple merge (không RRF)               & 72{,}8 & $-15{,}2$ \\

    \bottomrule
  \end{tabular}

Phân tích:
\begin{itemize}
  \item \textbf{RRF là thành phần quan trọng nhất:} Hợp nhất RRF đóng góp +23,2 pp so với
  chỉ dùng duyệt cây, và +15,2 pp so với simple merge. Không có RRF, mở rộng
  KG và Tầng 2 gần như không đóng góp.

  \item \textbf{Duyệt cây là backbone:} Tầng 1 alone (64,8\%) vượt xa Tầng 2 alone (33,0\%), xác
  nhận rằng tín hiệu cấu trúc đáng tin cậy hơn tín hiệu ngữ nghĩa cho văn bản pháp
  lý có cấu trúc.

  \item \textbf{Tầng 2 bổ trợ qua RRF:} Dual-Level alone yếu (33,0\%), nhưng khi kết hợp qua
  RRF đóng góp +15,0 pp so với Tree+KG, cho thấy giá trị bổ trợ thông qua RRF.

  \item \textbf{Variance thấp:} Tất cả cấu hình có variance $\pm$0,2--0,5 pp qua 3 lần chạy, xác nhận
  tính ổn định.
\end{itemize}

\subsection{Phân tích lỗi}
\label{subsec:phan-tich-loi}

Trong 31 câu hỏi bị bỏ sót (7,8\%), 27 thuộc lĩnh vực doanh nghiệp:
\begin{itemize}
  \item \textbf{Phụ thuộc xuyên nghị định (6 câu):} Câu hỏi yêu cầu kết hợp thông tin từ luật
  chính và nghị định hướng dẫn.

  \item \textbf{Ngoài kho tài liệu (4 câu):} Câu hỏi về hộ kinh doanh, nằm ngoài phạm vi kho tài
  liệu.

  \item \textbf{Đa điều khoản (14 câu):} Câu hỏi yêu cầu $>$2 điều khoản, chủ yếu liên kết bằng
  chuỗi suy luận pháp lý ngầm thay vì tham chiếu chéo rõ ràng.

  \item \textbf{Lỗi Loop 0 (3 câu):} Chọn sai tài liệu ở bước đầu tiên.
\end{itemize}

\section{Ứng dụng minh họa}
\label{sec:ung-dung-minh-hoa}

Để minh họa khả năng ứng dụng thực tế của hệ thống truy xuất, một ứng dụng chat-bot tra cứu quy định đào tạo đã được xây dựng dưới dạng giao diện web tương tác. Ứng dụng kết nối trực tiếp với pipeline truy xuất 3-Tier KG-RAG và sử dụng LLM để sinh câu trả lời theo phương pháp phân tích IRAC (Issue--Rule--Application--Conclusion), phục vụ trực tiếp nhu cầu tra cứu quy chế đào tạo sau đại học của học viên cao học.

\subsection{Kiến trúc ứng dụng}
\label{subsec:kientruc-ungdung}

Ứng dụng được xây dựng trên nền tảng Chainlit 2.0, một framework mã nguồn mở
cho giao diện chatbot hỗ trợ streaming. Kiến trúc gồm ba thành phần chính:

\begin{itemize}
  \item \textbf{Giao diện web (Chainlit):} Giao diện chat hỗ trợ streaming token-by-token, hiển
  thị trạng thái truy xuất theo thời gian thực và panel nguồn trích dẫn với bread-crumb vị trí trong văn bản.

  \item \textbf{Cầu nối bất đồng bộ (Async Bridge):} Pipeline truy xuất 3-Tier hoạt động đồng
  bộ (synchronous), trong khi Chainlit yêu cầu xử lý bất đồng bộ (asynchronous) để
  duy trì kết nối WebSocket. Một cầu nối sử dụng \texttt{threading.Thread} và \texttt{asyncio.Queue}
  chuyển tiếp sự kiện từ pipeline đồng bộ sang vòng lặp sự kiện bất đồng bộ, tránh
  chặn (block) giao diện trong suốt quá trình truy xuất (30--60 giây).

  \item \textbf{Pipeline truy xuất (3-Tier KG-RAG):} Pipeline sinh ba loại sự kiện: status (cập
  nhật trạng thái truy xuất), metadata (trích dẫn, reasoning trace, độ tin cậy) và
  token (nội dung câu trả lời streaming).
\end{itemize}

\subsection{Phương pháp sinh câu trả lời IRAC}
\label{subsec:phuongphap-irac}

IRAC (Issue--Rule--Application--Conclusion) là phương pháp phân tích có nguồn
gốc từ lĩnh vực pháp lý, được sử dụng rộng rãi trong đào tạo luật và tư vấn pháp luật.
Phương pháp này phù hợp cho bài toán hỏi đáp quy định vì đảm bảo câu trả lời có cấu
trúc rõ ràng, trích dẫn căn cứ cụ thể và kết luận dứt khoát. Bốn phần của IRAC gồm:

\begin{enumerate}
  \item \textbf{Issue (Vấn đề):} Xác định vấn đề cần giải đáp từ câu hỏi của người dùng.

  \item \textbf{Rule (Căn cứ quy chế):} Trích dẫn chính xác Điều, Khoản, Điểm của quy chế/quyết
  định liên quan từ ngữ cảnh đã truy xuất.

  \item \textbf{Application (Phân tích áp dụng):} Giải thích cụ thể quy định áp dụng vào tình
  huống như thế nào, chỉ ra mối liên hệ giữa quy chế và câu hỏi thực tế.

  \item \textbf{Conclusion (Kết luận):} Trả lời trực tiếp câu hỏi, tổng hợp các điều kiện hoặc quy
  trình.
\end{enumerate}

Hệ thống sử dụng prompt engineering để hướng dẫn LLM sinh câu trả lời theo cấu
trúc IRAC. Prompt được tùy biến theo miền ứng dụng, với hai điểm khác biệt chính giữa
miền pháp luật và miền đào tạo:

\subsection{Các tính năng chính}
\label{subsec:tinh-nang-chinh}

\begin{enumerate}
  \item \textbf{Phản hồi trạng thái theo thời gian thực:} Trong quá trình truy xuất, hệ thống
  hiển thị trạng thái cập nhật trực tiếp qua ba giai đoạn: ``Đang phân tích câu hỏi'',
  ``Đang tìm kiếm quy định liên quan'' và ``Đang tổng hợp câu trả lời'' (Hình~\ref{fig:retrieving-state}).
  Thông báo tự động biến mất khi câu trả lời bắt đầu streaming. Điều này giải quyết
  vấn đề UX quan trọng khi thời gian truy xuất kéo dài 30--60 giây.

  \item \textbf{Streaming câu trả lời:} Câu trả lời được sinh theo phương pháp IRAC và stream-ing token-by-token đến giao diện, giúp người dùng bắt đầu đọc ngay khi LLM
  sinh văn bản mà không cần đợi toàn bộ câu trả lời (Hình~\ref{fig:irac-answer}).

  \item \textbf{Trích dẫn nguồn với breadcrumb:} Panel bên phải hiển thị nội dung gốc của các
  điều khoản được trích dẫn, với breadcrumb vị trí rõ ràng (tên văn bản, chương,
  điều) và đường phân cách giữa các nguồn. Người dùng có thể đối chiếu trực tiếp
  giữa tư vấn và nguyên văn quy định (Hình~\ref{fig:citation-panel}). Ngoài ra, lộ trình
  truy xuất 3 vòng lặp (Loop 0--2) cũng được hiển thị minh bạch để người dùng hiểu
  quá trình tìm kiếm của hệ thống (Hình~\ref{fig:reasoning-path}).

  \item \textbf{Hỗ trợ đa miền:} Ứng dụng hỗ trợ chuyển đổi giữa các miền (quy định đào
  tạo, pháp luật) thông qua biến môi trường cấu hình, sử dụng cùng giao diện và
  pipeline.
\end{enumerate}

\subsection{Ví dụ minh họa tiêu biểu}
\label{subsec:vi-du-minh-hoa}

Phần này trình bày các ví dụ cụ thể từ kết quả thử nghiệm trên miền quy định đào
tạo, minh họa điểm mạnh và hạn chế của hệ thống qua so sánh trực tiếp với các baseline.
Các câu hỏi được trích nguyên văn từ diễn đàn học viên cao học UIT.

\subsubsection*{Ví dụ 1: Truy xuất cross-document -- yêu cầu ngoại ngữ}

\textit{Câu hỏi:} ``Chứng chỉ TOEIC do các trường ĐH, CĐ trong nước cấp (không phải
của trung tâm khảo thí của ETS cấp) có được công nhận anh văn đầu ra Cao học không
ạ?''

\textit{Gold articles:} Điều 6 QĐ 1393 (ĐHQG-HCM, yêu cầu ngoại ngữ đầu ra) và Điều 9
QĐ 270 (UIT, quy định bổ sung về chứng chỉ được chấp nhận).

\textit{Phân tích:} Câu hỏi yêu cầu kết hợp thông tin từ hai văn bản ở hai tầng quản lý
khác nhau: QĐ 1393 (ĐHQG-HCM) quy định yêu cầu chung, còn QĐ 270 (UIT) bổ
sung danh sách chứng chỉ cụ thể được chấp nhận. Tầng 1 chọn đúng cả hai tài liệu ở
Loop 0, sau đó Loop 1 thu hẹp vào các chương về ngoại ngữ. KG expansion bổ sung
Điều 25 (QĐ 270) qua cạnh references từ Điều 9. Hệ thống đạt MRR = 1,0 với cả hai
gold articles trong top 3. PageIndex tìm được Điều 6 nhưng bỏ sót Điều 9 của UIT (chỉ
tìm trong QĐ 1393 và QĐ 160). RAPTOR không tìm được cả hai, truy xuất phân tán
giữa các điều khoản không liên quan.

biết về quy trình đăng ký khác khóa. Tầng 1 chọn đúng chương về đào tạo trong cả hai
văn bản. Cả PageIndex và RAPTOR đều không truy xuất được Điều 8 hoặc Điều 11

\subsubsection*{Ví dụ 4: Trường hợp thất bại -- câu hỏi thiếu ngữ cảnh}

\textit{Câu hỏi:} ``Thầy/cô cho em hỏi, nếu như học ngành CNTT, sau này đăng ký học Tiến

\textit{Gold articles:} Điều 6 QĐ 1393 (điều kiện trình độ đầu vào) và Điều 9 QĐ 270 (yêu
cầu bổ sung).

\textit{Kết quả:} Hệ thống truy xuất Điều 10, 11 QĐ 1393 (chương trình Tiến sĩ) và Điều 13,
17 QĐ 270 (nghiên cứu sinh) trong top 5 -- đúng chủ đề Tiến sĩ nhưng sai điều khoản cụ
thể. PageIndex lại truy xuất đúng nhờ tìm kiếm rộng hơn trên QĐ 270.

\textit{Nguyên nhân:} Câu hỏi hỏi về ``Tiến sĩ'' nhưng gold articles nằm ở phần ``điều kiện
đầu vào Thạc sĩ'' (Điều 6, Điều 9). Nội dung hai điều khoản này bao gồm cả điều kiện
cho chương trình Tiến sĩ nhưng tiêu đề chương và tóm tắt chỉ đề cập ``Thạc sĩ'', khiến
Loop 1 chọn nhầm chương. Đây là hạn chế của duyệt cây khi tiêu đề chương không phản
ánh đầy đủ nội dung bên trong.
